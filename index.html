async function loadGallery() { const container = document.getElementById("galleryContainer"); if (!container) return; const { data, error } = await supabaseClient .from("gallery") .select("title, image_url") .order("id", { ascending: false }); if (error) { container.textContent = "Gallery लोड झाली नाही."; console.error(error); return; } container.replaceChildren(); if (!data || data.length === 0) { container.textContent = "अजून फोटो जोडलेले नाहीत."; return; } data.forEach(photo => { const card = document.createElement("div"); card.className = "card"; const img = document.createElement("img"); img.src = photo.image_url; img.alt = photo.title || "Team photo"; img.style.width = "100%"; img.style.borderRadius = "10px"; const title = document.createElement("h3"); title.textContent = photo.title || "Team Photo"; card.append(img, title); container.append(card); }); } loadGallery();
